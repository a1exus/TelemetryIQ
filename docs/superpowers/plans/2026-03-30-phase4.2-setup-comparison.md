@app.patch("/sessions/{session_id}")
async def patch_session(session_id: int, request: Request):
    from fastapi import HTTPException
    if _repo is None:
        raise HTTPException(status_code=503, detail="repository not ready")
    body = await request.json()
    if "notes" not in body:
        raise HTTPException(status_code=422, detail="missing 'notes' key")
    notes = body["notes"]
    if notes is not None and not isinstance(notes, str):
        raise HTTPException(status_code=422, detail="notes must be string or null")
    if notes == "":
        notes = None
    rows = await _repo.update_session_notes(session_id, notes)
    if rows == 0:
        raise HTTPException(status_code=404, detail="session not found")
    return {"ok": True}
```

function computeSetupDiff() {
  const el = document.getElementById('setup-diff');
  el.style.display = 'none';
  el.textContent = '';

  if (!state.lapA || !state.lapB) return;
  if (state.lapA.session_id === state.lapB.session_id) return;
  if (!state.lapA.frames.length || !state.lapB.frames.length) return;

  const fA = state.lapA.frames[0];
  const fB = state.lapB.frames[0];
  const diffs = [];

  // Gear ratios
  for (let i = 1; i <= 8; i++) {
    const key = 'gear' + i;
    const a = fA[key], b = fB[key];
    if (a == null && b == null) continue;
    if (a === 0 && b === 0) continue;
    if (a !== b) {
      diffs.push('G' + i + ': ' + Number(a).toFixed(3) + ' \u2192 ' + Number(b).toFixed(3));
    }
  }

  // Top speed setting
  if (fA.trans_top_speed != null && fB.trans_top_speed != null &&
      Math.round(fA.trans_top_speed) !== Math.round(fB.trans_top_speed)) {
    diffs.push('Top speed setting: ' + Math.round(fA.trans_top_speed) +
               ' \u2192 ' + Math.round(fB.trans_top_speed));
  }

  // Calculated max speed
  if (fA.calc_max_speed != null && fB.calc_max_speed != null &&
      Math.round(fA.calc_max_speed) !== Math.round(fB.calc_max_speed)) {
    diffs.push('Calc max speed: ' + Math.round(fA.calc_max_speed) +
               ' \u2192 ' + Math.round(fB.calc_max_speed));
  }

  if (!diffs.length) return;

  el.style.display = 'block';
  el.textContent = '\u2699 Auto-detected: ' + diffs.join(', ');
}
```

function startNoteEdit(session, notesEl) {
  const input = document.createElement('input');
  input.type = 'text';
  input.className = 'session-notes-input';
  input.value = session.notes || '';
  input.placeholder = 'e.g. front DF +5, softer rear springs';

  const save = async () => {
    const val = input.value.trim();
    const notes = val || null;
    try {
      const res = await fetch('/sessions/' + session.id, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ notes }),
      });
      if (res.ok) {
        session.notes = notes;
        if (state.lapA && state.lapA.session_id === session.id) state.lapA.session_notes = notes;
        if (state.lapB && state.lapB.session_id === session.id) state.lapB.session_notes = notes;
      }
    } catch (e) { /* network error */ }
    renderSidebar();
    if (state.lapA || state.lapB) updateCompareHeader();
  };

  const cancel = () => renderSidebar();

  input.addEventListener('keydown', (e) => {
    if (e.key === 'Enter') { e.preventDefault(); save(); }
    if (e.key === 'Escape') { e.preventDefault(); cancel(); }
  });
  input.addEventListener('blur', save);

  notesEl.replaceWith(input);
  input.focus();
  input.select();
}
